let objectionRules = {}; // 1. تحميل قواعد الاعتراضات تلقائياً عند فتح التطبيق fetch('./objection_rules.json') .then(response => response.json()) .then(data => { objectionRules = data; console.log("تم تحميل ملف القواعد بنجاح:", objectionRules); }) .catch(err => console.error("خطأ في قراءة ملف JSON:", err)); // 2. إعداد الميكروفون والتعرف على الصوت const SpeechRecognition = window.SpeechRecognition || window.webkitSpeechRecognition; if (SpeechRecognition) { const recognition = new SpeechRecognition(); recognition.lang = 'ar-SA'; // لغة التقاط الصوت العربية recognition.continuous = false; recognition.interimResults = false; const recordBtn = document.getElementById('recordBtn'); // تأكد من وجود زر بهذا الـ id recordBtn.addEventListener('click', () => { recognition.start(); console.log("جاري الاستماع للعميل..."); }); recognition.onresult = (event) => { const text = event.results[0][0].transcript.toLowerCase(); console.log("النص الملتقط:", text); matchObjection(text); }; recognition.onerror = (event) => { console.error("خطأ الميكروفون:", event.error); }; } // 3. مطابقة الاعتراض وإظهار الرد الميداني function matchObjection(spokenText) { let responseText = "لم يتم العثور على رد مخصص لهذا الاعتراض."; if (spokenText.includes("بايسن") || spokenText.includes("bison")) { if (spokenText.includes("سعر") || spokenText.includes("غالي")) responseText = objectionRules.bison?.price; else if (spokenText.includes("مخزون") || spokenText.includes("نقص")) responseText = objectionRules.bison?.stock; else responseText = objectionRules.bison?.price || responseText; } else if (spokenText.includes("بيكر") || spokenText.includes("baker")) { if (spokenText.includes("جودة") || spokenText.includes("تغليف")) responseText = objectionRules.baker?.quality; else responseText = objectionRules.baker?.price || responseText; } else if (spokenText.includes("حصان") || spokenText.includes("شاي")) { responseText = objectionRules.horse_head_tea?.quality || objectionRules.horse_head_tea?.price; } // عرض الرد في الشاشة (تأكد من وجود عنصر يحمل id="responseBox") const responseBox = document.getElementById('responseBox'); if (responseBox) responseBox.innerText = responseText; }
